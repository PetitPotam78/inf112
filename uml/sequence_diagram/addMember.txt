@startuml addMember

skinparam sequenceMessageAlign center
skinparam responseMessageBelowArrow true

' ============================================================
' Participants
' ============================================================
actor "Utilisateur" as user
participant "sn : SocialNetwork" as sn
participant "m : Member" as m

== addMember ==

user -> sn : addMember(login, password, profile)
activate sn

    ' --- Validation des credentials (auto-appels) ---
    sn -> sn : checkCredentials(login, password)
    activate sn
        sn -> sn : checkString(login, 1, "invalid login")
        activate sn
        sn --> sn : void
        deactivate sn

        sn -> sn : checkString(password, 4, "invalid password")
        activate sn
        sn --> sn : void
        deactivate sn
    sn --> sn : void
    deactivate sn

    ' --- Validation du profil ---
    alt profile == null
        sn --> user : throw BadEntryException
    end

    ' --- Vérification d'unicité du login ---
    loop for each m in members
        sn -> m : hasSameLoginAs(login)
        activate m
        m --> sn : boolean
        deactivate m

        alt hasSameLoginAs == true
            sn --> user : throw MemberAlreadyExistsException
        end
    end

    ' --- Création et enregistrement du membre ---
    create "newMember : Member" as newMember
    sn -> newMember : Member(login, password, profile)
    activate newMember
    newMember --> sn : void
    deactivate newMember

sn --> user : void
deactivate sn

@enduml
