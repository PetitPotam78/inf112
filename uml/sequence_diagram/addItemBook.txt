@startuml addItemBook

skinparam sequenceMessageAlign center
skinparam responseMessageBelowArrow true

' ============================================================
' Participants
' ============================================================
actor "Utilisateur" as user
participant "sn : SocialNetwork" as sn
participant "m : Member" as m
participant "b : Book" as b

== addItemBook ==

user -> sn : addItemBook(login, password, title, kind, author, nbPages)
activate sn

    ' --- Validation des credentials (auto-appels) ---
    sn -> sn : checkCredentials(login, password)
    activate sn
        sn -> sn : checkString(login, 1, "invalid login")
        activate sn
        sn --> sn : void
        deactivate sn

        sn -> sn : checkString(password, 4, "invalid password")
        activate sn
        sn --> sn : void
        deactivate sn
    sn --> sn : void
    deactivate sn

    alt login ou password invalide
        sn --> user : throw BadEntryException
    end

    ' --- Validation des champs du livre (auto-appels) ---
    sn -> sn : checkBookFields(title, kind, author, nbPages)
    activate sn
        sn -> sn : checkString(title, 1, "invalid title")
        activate sn
        sn --> sn : void
        deactivate sn

        note right of sn
            Vérifie également :
            kind != null
            author != null
            nbPages > 0
        end note
    sn --> sn : void
    deactivate sn

    alt title/kind/author invalide ou nbPages <= 0
        sn --> user : throw BadEntryException
    end

    ' --- Authentification du membre ---
    sn -> sn : getAuthenticatedMember(login, password)
    activate sn
        loop for each m in members
            sn -> m : hasSameCredentials(login, password)
            activate m
            m --> sn : boolean
            deactivate m

            alt hasSameCredentials == true
                sn --> sn : return m
            end
        end

        alt aucun membre trouvé
            sn --> user : throw NotMemberException
        end
    deactivate sn

    ' --- Vérification d'unicité du titre ---
    loop for each b in books
        sn -> b : hasSameTitleAs(title)
        activate b
        b --> sn : boolean
        deactivate b

        alt hasSameTitleAs == true
            sn --> user : throw ItemBookAlreadyExistsException
        end
    end

    ' --- Création et enregistrement du livre ---
    create "newBook : Book" as newBook
    sn -> newBook : Book(title, kind, author, nbPages)
    activate newBook
    newBook --> sn : void
    deactivate newBook

    note over sn : books.add(newBook)

sn --> user : void
deactivate sn

@enduml
