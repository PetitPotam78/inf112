package "opinion" {

    interface ISocialNetwork {
        + nbMembers() : int
        + nbFilms() : int
        + nbBooks() : int
        + addMember(login : String, password : String, profile : String) : void
        + addItemFilm(login : String, password : String, title : String, kind : String, director : String, scenarist : String, duration : int) : void
        + addItemBook(login : String, password : String, title : String, kind : String, author : String, nbPages : int) : void
        + reviewItemFilm(login : String, password : String, title : String, mark : float, comment : String) : float
        + reviewItemBook(login : String, password : String, title : String, mark : float, comment : String) : float
        + consultItems(title : String) : LinkedList<String>
        + toString() : String
    }

    class SocialNetwork {
        + nbMembers() : int
        + nbFilms() : int
        + nbBooks() : int
        + addMember(login : String, password : String, profile : String) : void
        + addItemBook(login : String, password : String, title: String, kind : String, author : String, nbPages : int)
        + toString() : String
    }

    class Member {
        - login : String
        - password : String
        - profile : String
        + getLogin() : String
        + getPassword() : String
        + getProfile() : String
    }

    class Book {
        - title : String
        - kind : String
        - author : String
        - nbPages : int
        + getTitle() : String
        + getKind() : String
        + getAuthor() : String
        + getNbPages() : int
    }

    class Review {
        - mark : float
        - comment : String
    }

}

package "exceptions" {
    class BadEntryException
    class MemberAlreadyExistsException
    class ItemBookAlreadyExistsException
}

SocialNetwork ..|> ISocialNetwork
SocialNetwork "0..1" --> "0..*" Member : members
SocialNetwork "0..1" --> "0..*" Book : books

Book "0..1" --> "0..*" Review : reviews

Review "0..*" --> "1" Member : author

SocialNetwork ..> BadEntryException : throws
SocialNetwork ..> MemberAlreadyExistsException : throws
SocialNetwork ..> ItemBookAlreadyExistsException : throws

@enduml
